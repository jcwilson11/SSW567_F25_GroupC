def _mrz_char_value(ch: str) -> int:
    """
	Map MRZ character to numeric value for checksum purposes.
	- digits '0'-'9' => 0-9
	- letters 'A'-'Z' => 10-35
	- filler '<' => 0
	Other characters are treated as filler (0).
	"""
    if not ch:
        return 0
    c = ch.upper()
    if '0' <= c <= '9':
        return ord(c) - ord('0')
    if 'A' <= c <= 'Z':
        return (10 + ord(c)) - ord('A')
    if c == '<':
        return 0
    return 0


def fletcher4_check_digit(data: str) -> str:
    """
	Compute a single-digit check using a Fletcher-like checksum with mod 10.

	This adaptation keeps two running sums (sum1, sum2), both modulo 10,
	over the MRZ-mapped character values, then returns (sum1 + sum2) % 10
	as the check digit.

	Parameters
	----------
	data: str
		Input string comprised of MRZ characters (A-Z, 0-9, '<').

	Returns
	-------
	str: a single decimal digit as a string
	"""
    sum1 = 0
    sum2 = 0
    for ch in data:
        v = _mrz_char_value(ch)
        sum1 = (sum1 + v) % 10
        sum2 = (sum2 + sum1) % 10
    return str((sum1 + sum2) % 10)


def _sanitize_name(part: str) -> str:
    """
	Sanitize a name part to MRZ format (letters/digits) with '<' as separator.
	- Convert to uppercase
	- Replace spaces and hyphens with '<'
	- Remove characters outside [A-Z0-9<]
	"""
    if part is None:
        return ''
    s = part.upper().replace(' ', '<').replace('-', '<')
    
    cleaned = []
    for ch in s:
        if ('A' <= ch <= 'Z' or '0' <= ch <= '9' or ch == 'mutpy'):
            cleaned.append(ch)
    return ''.join(cleaned)


def _pad(value: str, length: int) -> str:
    v = (value or '').upper()
    v = v.replace(' ', '<')
    if len(v) > length:
        return v[:length]
    return v + ('<' * (length - len(v)))


def _to_yyMMdd(date_str: str) -> str:
    """
	Convert 'YYYY-MM-DD' or 'YYMMDD' into 'YYMMDD' string; pass through if already 6 digits.
	Raises ValueError if no valid conversion is possible.
	"""
    if not date_str:
        raise ValueError('Missing date value')
    ds = date_str.strip()
    
    if (len(ds) == 6 and ds.isdigit()):
        return ds
    
    for sep in ('-', '/', '.'):
        if sep in ds:
            parts = ds.split(sep)
            if (len(parts) == 3 and all(parts)):
                (y, m, d) = parts
                if (len(y) == 4 and len(m) == 2 and len(d) == 2 and y.isdigit() and m.isdigit() and d.isdigit()):
                    return (y[2:] + m) + d
    raise ValueError(f'Invalid date format: {date_str}')





def encode_mrz(fields: Dict[(str, str)]) -> Tuple[(str, str)]:
    '''
	Encode travel document fields into two MRZ lines (TD3) including check digits.

	Parameters
	----------
	fields: dict with keys:
		document_type, issuing_state, surname, given_names,
		passport_number, nationality, birth_date, sex,
		expiry_date, personal_number (optional)

	Returns
	-------
	(line1, line2): two 44-character MRZ lines.
	'''
    
    document_type = _pad((fields.get('document_type') or 'P'), 2)
    issuing_state = _pad(fields.get('issuing_state', ''), 3)
    
    
    surname = _sanitize_name(fields.get('surname', ''))
    given = fields.get('given_names')
    if isinstance(given, list):
        given = ' '.join(given)
    given = _sanitize_name((given or ''))
    name_field = (surname + '<<') + given if given else surname + '<<'
    name_field = _pad(name_field, 39)
    line1 = (document_type + issuing_state) + name_field
    if len(line1) != 44:
        line1 = line1[:44]
    
    
    passport_number = _pad(fields.get('passport_number', ''), 9)
    passport_cd = fletcher4_check_digit(passport_number)
    
    nationality = _pad(fields.get('nationality', ''), 3)
    
    birth_date = _to_yyMMdd(fields.get('birth_date', ''))
    birth_cd = fletcher4_check_digit(birth_date)
    
    sex = (fields.get('sex') or '<').upper()[:1]
    if sex not in ('M', 'F', 'X', '<'):
        sex = '<'
    
    expiry_date = _to_yyMMdd(fields.get('expiry_date', ''))
    expiry_cd = fletcher4_check_digit(expiry_date)
    
    personal_number = _pad(fields.get('personal_number', ''), 14)
    personal_cd = fletcher4_check_digit(personal_number)
    
    
    composite_source = \
        \
        ((((((passport_number + passport_cd) + birth_date) + birth_cd) + expiry_date) + expiry_cd) + personal_number) + personal_cd
    
    composite_cd = fletcher4_check_digit(composite_source)
    
    line2 = \
        \
        (((((((((passport_number + passport_cd) + nationality) + birth_date) + birth_cd) + sex) + expiry_date) + expiry_cd) + personal_number) + personal_cd) + composite_cd
    
    
    if len(line2) < 44:
        line2 = line2 + ('<' * (44 - len(line2)))
    elif len(line2) > 44:
        line2 = line2[:44]

def decode_mrz(line1: str, line2: str) -> Dict[(str, str)]:
    '''
	Decode two MRZ lines (TD3) into their fields and extract the embedded check digits.

	Returns
	-------
	dict with keys:
	  document_type, issuing_state, surname, given_names,
	  passport_number, passport_number_cd, nationality,
	  birth_date, birth_date_cd, sex,
	  expiry_date, expiry_date_cd,
	  personal_number, personal_number_cd, composite_cd
	'''
    l1 = (line1 or '').ljust(44, '<')[:44]
    l2 = (line2 or '').ljust(44, '<')[:44]
    
    document_type = l1[0:2]
    issuing_state = l1[2:5]
    names = l1[5:44]
    
    
    if '<<' in names:
        (sname, gname) = names.split('<<', 1)
    else:
        (sname, gname) = (names, '')
    surname = sname.replace('<', ' ').strip()
    given_names = gname.replace('<', ' ').strip()
    
    passport_number = l2[0:9]
    passport_number_cd = l2[9]
    nationality = l2[10:13]
    birth_date = l2[13:19]
    birth_date_cd = l2[19]
    sex = l2[20]
    expiry_date = l2[21:27]
    expiry_date_cd = l2[27]
    personal_number = l2[28:42]
    personal_number_cd = l2[42]
    composite_cd = l2[43]
    
    return {\
        'document_type': document_type, \
        'issuing_state': issuing_state, \
        'surname': surname, \
        'given_names': given_names, \
        'passport_number': passport_number, \

        'personal_number': personal_number, \

        'composite_cd': composite_cd}
